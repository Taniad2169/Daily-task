Deploy > Manage deployments > pencil (edit) icon on your existing * deployment > Version: "New version" > Deploy. * (Your Daily Task app already has this exact deployment URL wired * in - AKfycbwfAVJ2lzemCMFroJCEZHJLJ45VXpoMoLbL6EOZK2cPlBxg5PlANoLObLD2p3k4AEVAsA - * so this is an UPDATE to that URL, not a new one. Nothing in the * HTML file needs to change.) * 5. IMPORTANT one-time step: the first time you run/deploy this after * adding the Calendar part below, Google will ask you to authorize * the script to access your Calendar (a permissions popup). You'll * need to click through and allow it once - after that it works * automatically every time. * * This does TWO things per lead: * 1. Writes a row into the sheet (same as before): A=Lead/Business, * B=Contact Name, C=Phone, D=Email, E=Website, F=Action Needed, * G/H/I=Report Prepared?/Email Drafted?/Email Sent? (yours to fill * in by hand), J=Notes. * 2. If a date and time were picked in the Calendar button's popup, * ALSO creates a real 30-minute event on your Google Calendar at * that exact date/time - no separate script needed, no calendar * link needed. CalendarApp.getDefaultCalendar() automatically uses * whichever Google account this script is deployed under (yours), * so it just works. */ function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Sheet1') || SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; const data = JSON.parse(e.postData.contents); let actionNeeded = ''; let calendarResult = ''; // If a date/time was picked, create a REAL calendar event AND note it // in the sheet's "Action Needed" column. if (data.eventDate && data.eventTime) { actionNeeded = 'Scheduled: ' + data.eventDate + ' at ' + data.eventTime; try { const startTime = new Date(data.eventDate + ' ' + data.eventTime); const endTime = new Date(startTime.getTime() + 30 * 60 * 1000); // 30-minute event const title = 'Lead: ' + (data.name || 'Follow-up') + (data.phone ? ' (' + data.phone + ')' : ''); const descriptionLines = [ 'Email: ' + (data.email || 'N/A'), 'Website: ' + (data.url || 'N/A'), 'Notes: ' + (data.comment || 'N/A') ]; CalendarApp.getDefaultCalendar().createEvent( title, startTime, endTime, { description: descriptionLines.join('\n') } ); calendarResult = 'created'; } catch (calErr) { // If the calendar event fails for any reason (bad date format, // permissions not yet granted, etc.), the sheet row below still // gets written - a calendar hiccup should never lose the lead. calendarResult = 'failed: ' + String(calErr); } } const row = [ '', // A - Lead / Business (no separate business field in this form) data.name || '', // B - Contact Name data.phone || '', // C - Phone data.email || '', // D - Email data.url || '', // E - Website actionNeeded, // F - Action Needed '', // G - Report Prepared? (you track by hand) '', // H - Email Drafted? (you track by hand) '', // I - Email Sent? (you track by hand) data.comment || '' // J - Notes ]; sheet.appendRow(row); return ContentService.createTextOutput(JSON.stringify({ ok: true, calendar: calendarResult })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) })) .setMimeType(ContentService.MimeType.JSON); } }
